## 1. 数据模型

```sql
-- 1.1 积分余额:两个桶
--     free_balance = 免费所得(新手包/签到/邀请)。不过期;余额 ≥cap 时不再发签到。先被扣。
--     paid_balance = 充值所得(激活码)。不过期、无上限、不受 cap 影响。后被扣。
create table if not exists public.user_credits (
  user_id uuid primary key references auth.users(id) on delete cascade,
  free_balance integer not null default 0,
  paid_balance integer not null default 0,
  updated_at timestamptz not null default now()
);
alter table public.user_credits enable row level security;
create policy uc_read_own on public.user_credits for select using (user_id = auth.uid());
-- 无 insert/update/delete 策略 = 客户端只读

-- 1.2 积分流水(审计;正=入账,负=消耗)
create table if not exists public.credit_ledger (
  id bigint generated always as identity primary key,
  user_id uuid not null,
  delta integer not null,
  reason text not null,
    -- welcome_bonus / daily_checkin / referral_invitee / referral_inviter / redeem_code
    -- space_creation / unit_generation / ask_ai / refund
  idem_key text,
  balance_after integer,
  created_at timestamptz not null default now()
);
alter table public.credit_ledger enable row level security;
create unique index if not exists credit_ledger_idem
  on public.credit_ledger(user_id, idem_key) where idem_key is not null;
create index if not exists credit_ledger_user_time on public.credit_ledger(user_id, created_at desc);
create policy cl_read_own on public.credit_ledger for select using (user_id = auth.uid());

-- 1.3 每日签到(主键即防重复领取)
create table if not exists public.daily_checkins (
  user_id uuid not null references auth.users(id) on delete cascade,
  checkin_date date not null,            -- 按 UTC+8 计日
  credits_granted integer not null,      -- 实际发放(达上限时为 0)
  created_at timestamptz not null default now(),
  primary key (user_id, checkin_date)
);
alter table public.daily_checkins enable row level security;
create policy dc_read_own on public.daily_checkins for select using (user_id = auth.uid());

-- 1.4 激活码
create table if not exists public.credit_codes (
  code text primary key,
  credits integer not null,
  batch text,
  note text,                             -- 对账用:对应谁的哪笔转账
  redeemed_by uuid references auth.users(id),
  redeemed_at timestamptz,
  expires_at timestamptz,                -- 可空 = 永久有效
  active boolean not null default true,
  created_at timestamptz not null default now()
);
alter table public.credit_codes enable row level security;
-- ⚠️ 故意不建任何 RLS 策略 = 客户端完全不可读不可写。
--    若给了 select 策略,用户可直接拖库拿到全部未使用的码 —— 本表最大安全风险点。

-- 1.5 兑换失败尝试(防暴力猜码)
create table if not exists public.code_attempts (
  user_id uuid not null,
  attempted_at timestamptz not null default now()
);
create index if not exists code_attempts_idx on public.code_attempts(user_id, attempted_at desc);
alter table public.code_attempts enable row level security;   -- 无策略,客户端不可读
```

> **为什么新手包 150 进"充值桶"而不是免费桶**:免费桶上限 100,新手包 150 会超。放进 `paid_balance` 使它①不触发上限逻辑;②语义上就是"送你的、永久属于你的";③新用户仍可立即签到(免费桶为 0 < 100)。**扣费先扣免费桶,所以新手包会被最后花掉——这是对的。**

## 3. 积分核心函数

```sql
-- 3.1 统一入账口(新手包 / 签到 / 邀请 / 激活码都走这里;将来接真实支付也走这里)
create or replace function public.grant_credits(
  p_user uuid, p_amount int, p_reason text, p_idem text, p_to_paid boolean default true)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_prev jsonb;
begin
  if p_amount <= 0 then return jsonb_build_object('ok', false, 'reason', 'invalid_amount'); end if;

  if p_idem is not null then                                     -- 幂等
    select jsonb_build_object('ok', true, 'reason', 'idempotent_replay', 'balance', balance_after)
      into v_prev from public.credit_ledger
     where user_id = p_user and idem_key = p_idem limit 1;
    if v_prev is not null then return v_prev; end if;
  end if;

  insert into public.user_credits(user_id) values (p_user) on conflict (user_id) do nothing;

  if p_to_paid then
    update public.user_credits set paid_balance = paid_balance + p_amount,
           updated_at = now() where user_id = p_user;
  else
    update public.user_credits set free_balance = free_balance + p_amount,
           updated_at = now() where user_id = p_user;
  end if;

  insert into public.credit_ledger(user_id, delta, reason, idem_key, balance_after)
  select p_user, p_amount, p_reason, p_idem, free_balance + paid_balance
    from public.user_credits where user_id = p_user;

  return jsonb_build_object('ok', true);
end $$;
-- 仅供其他 SECURITY DEFINER 函数与 service role 调用,不授权给 authenticated
revoke all on function public.grant_credits(uuid,int,text,text,boolean) from public, authenticated;

-- 3.2 每日签到(主键防重领 + 上限门槛防囤积)
create or replace function public.claim_daily_credits()
returns jsonb language plpgsql security definer set search_path = public as $$
declare
  v_user uuid := auth.uid();
  v_today date := (now() at time zone 'Asia/Shanghai')::date;
  v_amount int; v_cap int; f int; v_give int;
begin
  if v_user is null then return jsonb_build_object('ok', false, 'reason', 'not_authenticated'); end if;

  select daily_checkin_credits, free_balance_cap into v_amount, v_cap
    from public.billing_config where id;

  insert into public.user_credits(user_id) values (v_user) on conflict (user_id) do nothing;
  select free_balance into f from public.user_credits where user_id = v_user for update;

  -- 上限门槛:免费桶余额 ≥ cap 则签到成功但发 0(不囤积 = 不产生成本)
  v_give := case when coalesce(f,0) >= v_cap then 0 else v_amount end;

  begin
    insert into public.daily_checkins(user_id, checkin_date, credits_granted)
    values (v_user, v_today, v_give);
  exception when unique_violation then
    return jsonb_build_object('ok', false, 'reason', 'already_claimed_today');
  end;

  if v_give > 0 then
    update public.user_credits set free_balance = f + v_give, updated_at = now()
     where user_id = v_user;
    insert into public.credit_ledger(user_id, delta, reason, idem_key, balance_after)
    select v_user, v_give, 'daily_checkin', 'checkin:'||v_today::text,
           free_balance + paid_balance from public.user_credits where user_id = v_user;
  end if;

  return jsonb_build_object('ok', true, 'granted', v_give, 'capped', (v_give = 0));
end $$;
revoke all on function public.claim_daily_credits() from public;
grant execute on function public.claim_daily_credits() to authenticated;

-- 3.3 扣费(原子、行锁防并发透支、idem 防重扣;先扣免费桶后扣充值桶)
create or replace function public.spend_credits(p_cost int, p_reason text, p_idem text default null)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_user uuid := auth.uid(); f int; p int; from_f int; v_prev jsonb;
begin
  if v_user is null then return jsonb_build_object('ok', false, 'reason', 'not_authenticated'); end if;

  if p_idem is not null then
    select jsonb_build_object('ok', true, 'reason', 'idempotent_replay', 'balance', balance_after)
      into v_prev from public.credit_ledger
     where user_id = v_user and idem_key = p_idem limit 1;
    if v_prev is not null then return v_prev; end if;
  end if;

  select free_balance, paid_balance into f, p
    from public.user_credits where user_id = v_user for update;

  if coalesce(f,0) + coalesce(p,0) < p_cost then
    return jsonb_build_object('ok', false, 'reason', 'insufficient_credits',
                              'balance', coalesce(f,0)+coalesce(p,0), 'needed', p_cost);
  end if;

  from_f := least(f, p_cost);                      -- 先扣免费桶
  update public.user_credits
     set free_balance = f - from_f,
         paid_balance = p - (p_cost - from_f),     -- 不够再扣充值桶
         updated_at = now()
   where user_id = v_user;

  insert into public.credit_ledger(user_id, delta, reason, idem_key, balance_after)
  values (v_user, -p_cost, p_reason, p_idem, (f + p - p_cost));

  return jsonb_build_object('ok', true, 'balance', f + p - p_cost);
end $$;
revoke all on function public.spend_credits(int,text,text) from public;
grant execute on function public.spend_credits(int,text,text) to authenticated;
```

**新用户初始化**(扩展既有 `handle_new_user`,只在 `return new` 前追加,不动它已有的 `referral_code` 等逻辑):

## 4. 激活码

### 4.1 批量生成(创始人在 Supabase SQL Editor 以 service role 跑)

```sql
create or replace function public.generate_credit_codes(
  p_count int, p_credits int, p_batch text default null, p_note text default null)
returns table(code text) language plpgsql security definer set search_path = public as $$
declare i int; v_code text; v_seg text;
  v_alphabet text := 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';   -- 剔除易混的 0 O 1 I L
begin
  for i in 1..p_count loop
    loop
      v_code := 'KV';
      for j in 1..3 loop
        select 'KV' into v_seg;                            -- 占位,下面重算
        v_seg := (select string_agg(
                    substr(v_alphabet, 1+floor(random()*length(v_alphabet))::int, 1), '')
                  from generate_series(1,4));
        v_code := v_code || '-' || v_seg;
      end loop;
      begin
        insert into public.credit_codes(code, credits, batch, note)
        values (v_code, p_credits, p_batch, p_note);
        exit;
      exception when unique_violation then continue;       -- 撞码重生成
      end;
    end loop;
    code := v_code; return next;
  end loop;
end $$;
revoke all on function public.generate_credit_codes(int,int,text,text) from public, authenticated;
```

> 码格式 `KV-XXXX-XXXX-XXXX`(12 位有效字符,32 字母表 ≈ 60 bit 熵),便于在微信里手打。
> **用法**:`select * from public.generate_credit_codes(10, 1000, '2026-08', '¥29.9 档');`
> **对账**:发码时把买家信息写进 `note`,或按 `batch` 分批统计。

### 4.2 兑换 RPC(用户端唯一入口)

```sql
create or replace function public.redeem_credit_code(p_code text)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_user uuid := auth.uid(); v_credits int; v_fails int; v_norm text;
begin
  if v_user is null then return jsonb_build_object('ok', false, 'reason', 'not_authenticated'); end if;
  v_norm := upper(trim(p_code));

  -- 限速:10 分钟内失败 5 次则锁定,防暴力猜码
  select count(*) into v_fails from public.code_attempts
   where user_id = v_user and attempted_at > now() - interval '10 minutes';
  if v_fails >= 5 then return jsonb_build_object('ok', false, 'reason', 'too_many_attempts'); end if;

  -- 原子核销:UPDATE ... WHERE redeemed_by is null 即并发守卫
  update public.credit_codes
     set redeemed_by = v_user, redeemed_at = now()
   where code = v_norm
     and redeemed_by is null
     and active
     and (expires_at is null or expires_at > now())
  returning credits into v_credits;

  if v_credits is null then
    insert into public.code_attempts(user_id) values (v_user);
    -- ⚠️ 统一错误:不区分"不存在 / 已使用 / 已过期",否则返回值可被用来探测码是否存在
    return jsonb_build_object('ok', false, 'reason', 'invalid_code');
  end if;

  perform public.grant_credits(v_user, v_credits, 'redeem_code', 'code:'||v_norm, true);  -- 进充值桶
  return jsonb_build_object('ok', true, 'credits', v_credits);
end $$;
revoke all on function public.redeem_credit_code(text) from public;
grant execute on function public.redeem_credit_code(text) to authenticated;
```

## 5. 空间:改为花积分,额度模型退化为防滥用闸

**语义变更**:空间不再由额度控制,而是"花 50 积分开一个学习目标"。四分量额度模型的收费意义消失。

-- 5.1 硬上限改读配置常量(纯防滥用)
create or replace function public.enforce_space_limit()
returns trigger language plpgsql security definer set search_path = public as $$
declare v_cap int;
begin
  select space_hard_cap into v_cap from public.billing_config where id;
  if (select count(*) from public.spaces where user_id = new.user_id) >= coalesce(v_cap, 50) then
    raise exception 'SPACE_LIMIT_REACHED';
  end if;
  return new;
end; $$;

### 5.3 建空间扣费:原子 + 前置告知(体验关键)

建空间流程 = 新建 → **访谈(多轮)** → 理解确认 → 生成路径。LLM 成本从第一轮访谈就开始发生。

```sql
create or replace function public.create_space_with_credits(p_title text, p_idem text)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_user uuid := auth.uid(); v_cost int; v_spend jsonb; v_space uuid;
begin
  if v_user is null then return jsonb_build_object('ok', false, 'reason', 'not_authenticated'); end if;
  select cost_space_creation into v_cost from public.billing_config where id;

  v_spend := public.spend_credits(v_cost, 'space_creation', p_idem);
  if not (v_spend->>'ok')::boolean then return v_spend; end if;   -- 积分不足直接返回,不建空间

  insert into public.spaces(user_id, title) values (v_user, p_title) returning id into v_space;
  -- 若此处触发 SPACE_LIMIT_REACHED 或任何异常,整个事务回滚,积分自动退回

  return jsonb_build_object('ok', true, 'space_id', v_space);
end $$;
revoke all on function public.create_space_with_credits(text,text) from public;
grant execute on function public.create_space_with_credits(text,text) to authenticated;
```

**规则(必须落到 UI,见 §8)**:
- 在**访谈开始前**扣费,并明确告知"将消耗 50 积分";
- **绝不能**访谈 10 轮后才提示积分不够;
- **中途放弃不退**(成本已发生),但这必须在开始前讲清楚;
- `p_idem` 用客户端生成的一次性 uuid,防止重试重扣。

## 6. 计量接线

调 LLM **之前**先扣费,`ok:false` 则**不调 LLM**,回 `CREDITS_EXHAUSTED` 给前端触发积分墙。

| 动作 | 扣费 | idem_key |
|---|---|---|
| **建空间**(访谈 + 摘要 + 路径) | `create_space_with_credits` 内部扣 50 | 客户端一次性 uuid |
| 单元生成 | `spend_credits(cost_unit_generation,'unit_generation',<unit_id>)` | 单元 id |
| 划词问 AI | `spend_credits(cost_ask_ai,'ask_ai',<question_id>)` | 提问 id |
| 阅读 / 提问记录 / 后台档案归纳 | **不扣** | — |

**LLM 调用失败**:建空间(50)与单元生成(10)属大额,失败即退积分(`grant_credits(..., 'refund', ...)`);问 AI(1)不退,打日志即可。扣费成功后照旧写 `llm_usage`,成本可见性不变。

## 7. 邀请奖励:从送空间改为送积分

改造既有 `apply_referral`,**保留全部原有守卫**(未登录 / 已被邀请 / 无效码 / 自邀 / 幂等),只把奖励换掉:

```sql
create or replace function public.apply_referral(p_code text)
returns jsonb language plpgsql security definer set search_path = public as $$
declare
  v_invitee uuid := auth.uid(); v_inviter uuid; v_already uuid;
  v_amount int; v_max int; v_times int;
begin
  if v_invitee is null then return jsonb_build_object('applied', false, 'reason', 'not_authenticated'); end if;

  select invited_by into v_already from public.app_users where id = v_invitee;
  if v_already is not null then return jsonb_build_object('applied', false, 'reason', 'already_referred'); end if;

  select id into v_inviter from public.app_users where referral_code = upper(trim(p_code));
  if v_inviter is null then return jsonb_build_object('applied', false, 'reason', 'invalid_code'); end if;
  if v_inviter = v_invitee then return jsonb_build_object('applied', false, 'reason', 'self_referral'); end if;

  update public.app_users set invited_by = v_inviter
   where id = v_invitee and invited_by is null;          -- race 守卫
  if not found then return jsonb_build_object('applied', false, 'reason', 'already_referred'); end if;

  select referral_credits, referral_inviter_max_times into v_amount, v_max
    from public.billing_config where id;

  -- 被邀请方:一次性 +50(进免费桶)
  perform public.grant_credits(v_invitee, v_amount, 'referral_invitee',
                               'ref_in:'||v_invitee::text, false);

  -- 邀请方:+50,累计封顶 5 次(按流水计数,可审计)
  select count(*) into v_times from public.credit_ledger
   where user_id = v_inviter and reason = 'referral_inviter';
  if v_times < v_max then
    perform public.grant_credits(v_inviter, v_amount, 'referral_inviter',
                                 'ref_out:'||v_invitee::text, false);
  end if;

  return jsonb_build_object('applied', true, 'credits', v_amount);
end $$;
```

> 两个 reason 刻意分开(`referral_invitee` / `referral_inviter`),否则邀请方的封顶计数会把它自己被邀请时拿的那 50 也算进去。
> 奖励进**免费桶**(可能超过 100 上限——**这是允许的**,上限只管签到发放)。
> **文案纪律**:这是配额钩子与付费漏斗,K 结构性小于 1,**不要包装成裂变/病毒增长**。

## 8. 前端(按《交互原型 · 纯积分制 Web V2》实现,已验收)

Web 与 App 一致。**所有数字取自 `billing_config_public`,不写死。**

**① 首页与签到**
- 顶部**积分余额芯片**:显示总积分(`free + paid`),低于 10 变暖红。**不出现"本月""刷新"等周期字样。**
- **签到组件**三态,文案必须区分清楚:
  - 未签到 → "今日签到 +5";
  - 已签到 → "今天已签到 / 明天再来 +5。免费积分不会过期";
  - **免费桶已达上限** → "免费额度已满(100),先用掉一些再来签到。充值额度不受影响" —— 按钮仍可点,点后记为已签、发 0。**这一态最容易让用户困惑,文案要诚实,不能假装发放。**
- "新建学习空间"卡片上标 **−50 积分**;继续学习区的按钮标 **−10 / −1**。
- 底部一行说明:积分只花在建空间 50 / 生成单元 10 / 问 AI 1,阅读与提问记录不花积分。

**② 建空间扣费确认(独立一步,不可省)**
- 一张明细卡:访谈 / 理解摘要 / 路径生成 → **合计 −50**,并列出当前余额与完成后剩余。
- 一条醒目警示:**"积分在访谈开始前扣除。中途退出不退还。"**
- 余额不足时,主按钮直接变为 **"积分不足(还差 X)"并禁用**,同屏给出"回首页签到"的出路。**不允许进入访谈后才发现不够。**

**③ 积分墙(唯一一种墙,三个触发点共用)**
- 标题按触发动作变化(建空间 / 生成单元 / 问 AI)。
- 必带一句:**"✓ 已生成的内容、提问记录照常看,功能没被锁"**。
- 三个出口,顺序固定:**兑换激活码(主) / 加微信充值 / 明天签到 +5**。
- 第三个出口不可省——它告诉用户不充值也能继续,只是慢一点,避免墙显得像勒索。

**④ 兑换激活码页**(「我的」里固定入口,不只在墙里)
- 输入框 + 兑换按钮;成功提示到账数量并即时刷新余额。
- 说明:**充值积分永不过期,不受免费额度上限影响**。
- 失败提示统一为 **"激活码无效或已被使用"**(与后端一致,不区分原因)。
- 附"还没有码?"引导:加微信 → 说档位 → 转账 → 发码。

**⑤ 积分流水页**
- 读 `credit_ledger`,倒序;每行:来源/去向、说明、增减、余额。
- 底部说明:**消耗时先扣免费额度、后扣充值额度,你花钱买的永远最后被动用。**
- 这一页是信任基建,**别省**——它能消掉绝大部分"我的积分怎么少了"的私信。

**⑥ 积分说明页 `/credits`(替代原定价页,公开可访问、SSR/SSG)**
- 主标题:功能全部开放,只有学多少的区别。
- 三张表:积分花在哪 / 积分从哪来 / 充值档位(¥9.9-300 / ¥29.9-1000 / ¥68-2500)。
- **必须公开写出:"一个学习目标从头走完 ≈ 340 积分"**(= 建空间 50 + 约 24 单元 240 + 追问 50)。用户知道这个数才知道该充哪档,透明本身就是转化工具。
- 充值流程 + 微信二维码;坦白说明"目前由创始人手工处理,通常几小时内"。

## 9. 埋点:别让签到污染核心指标(必做)

每日签到会把"为 5 积分点一下就走"的用户计入回访。必须分开统计:

## 11. 验收标准(重点:试着弄坏它)

**积分与两桶**
- [ ] 新用户注册即:`free_balance=0`、`paid_balance=150`,流水有一条 `welcome_bonus`
- [ ] **新用户能顺利建出第一个空间**(150 ≥ 50)并生成若干单元 —— onboarding 不断
- [ ] 扣费**先扣免费桶后扣充值桶**:免费桶 30、充值桶 100 时扣 50 → 免费桶 0、充值桶 80
- [ ] **不存在任何月度重置**:把系统时间跨月或等待跨月,余额纹丝不动
- [ ] 客户端直接 `update user_credits set paid_balance=99999`(自己)→ **被拒**

**签到**
- [ ] 同日重复签到 → `already_claimed_today`,不重发
- [ ] 免费桶 ≥100 时签到 → 发 0 但**仍记为已签到**,`capped=true`,UI 文案诚实
- [ ] 免费桶 98 时签到 → 正常发 5(变 103,**允许超过上限**,上限只管"发不发",不截断)
- [ ] 跨 UTC+8 零点后可再签一次

**激活码(安全重点)**
- [ ] 客户端 `select * from credit_codes` → **返回空/被拒**(拖不到未使用的码)
- [ ] 有效码兑换 → 进**充值桶**,`redeemed_by`/`redeemed_at` 被写上
- [ ] 同一个码兑换第二次 → 拒绝;**两账号同时抢同一个码 → 只有一个成功**(原子核销)
- [ ] 乱输码与已用码 → **返回完全相同的 `invalid_code`**(不可被用来探测码是否存在)
- [ ] 10 分钟内连错 5 次 → `too_many_attempts`
- [ ] `generate_credit_codes` 用普通登录用户调用 → **被拒**

**建空间与扣费**
- [ ] 余额 40 时点新建空间 → **在进入访谈前就被拦**,按钮显示"积分不足(还差 10)"
- [ ] 建空间过程中触发 `SPACE_LIMIT_REACHED` 或任何异常 → **积分回滚**,不出现"扣了积分没建成"
- [ ] 同一 `p_idem` 重试建空间 → 只扣一次
- [ ] **并发**:两标签页同时生成单元 → 无负余额(行锁生效)
- [ ] 单元生成时 LLM 失败 → 积分退回,流水有 `refund`
- [ ] 阅读 / 提问记录 / 后台档案归纳 → 扣 0
- [ ] 建到第 50 个空间可、第 51 个被服务端触发器拦(绕过客户端也拦)

**邀请**
- [ ] 用有效邀请码 → 双方各 +50,进免费桶,流水 reason 分别为 `referral_invitee` / `referral_inviter`
- [ ] 自邀 / 重复被邀 → 拒绝,积分不变
- [ ] 同一邀请方成功邀请 6 人 → **第 6 人只有被邀方拿到 50,邀请方停在 250**(封顶)
- [ ] 邀请方自己也被别人邀请过 → 它那条 `referral_invitee` **不计入**封顶计数

**通用**
- [ ] 两账号交叉:看不到、改不了对方的积分 / 流水 / 签到 / 兑换记录
- [ ] 客户端 bundle 内无 service role key;所有积分写入在服务端
- [ ] 前端所有数字来自 `billing_config_public`;改配置后前后端同时生效

## 12. 明确不做(范围纪律)

- ✗ **不做功能墙** —— 所有功能对所有人开放,只有积分多少的区别。
- ✗ **不做会员 / 订阅 / 分层**(单一货币)。
- ✗ **不做在线支付**(微信支付 / Paddle 均不接;充值 = 线下转账 + 激活码)。
- ✗ **不做月度发放与重置**(已从设计中删除,别"顺手"加回来)。
- ✗ 不做连续签到加成 / 补签 / 积分转赠 / 退款自助。
- ✗ 不做空间包、积分包 SKU(SKU 只存在于线下报价单,系统里只有"激活码面额")。
- ✗ 不改既有内容 / 学习功能。
- ✗ 客户端不放任何密钥;积分判断与写入只在服务端。
- ✗ 不做后端迁移与备案相关改造(合规敞口见《策划 V3》§1)。
